cmake_minimum_required(VERSION 3.20)
project(TargetSC VERSION 0.5.0 LANGUAGES CXX)
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
    message(FATAL_ERROR "TargetSC requires a 32-bit x86 compiler")
endif()
add_library(_TargetSC SHARED TargetSC.cpp StrokeGeometry.h PropertyGeometry.h exports.def)
set_target_properties(_TargetSC PROPERTIES PREFIX "" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/../libs")
target_compile_features(_TargetSC PRIVATE cxx_std_17)
target_link_libraries(_TargetSC PRIVATE d3d8)
if(MSVC)
    target_compile_options(_TargetSC PRIVATE /W4 /EHsc)
else()
    target_compile_options(_TargetSC PRIVATE -Wall -Wextra -Wpedantic -Werror)
    target_link_options(_TargetSC PRIVATE -static -static-libgcc -static-libstdc++)
endif()
